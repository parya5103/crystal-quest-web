cmake_minimum_required(VERSION 3.16)
project(CrystalQuest LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(COCOS2DX_ROOT_PATH ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/cocos2d-x)
if(NOT EXISTS "${COCOS2DX_ROOT_PATH}/cocos/CMakeLists.txt")
  message(FATAL_ERROR "Cocos2d-x v4 source missing. Run scripts/bootstrap-native-engine.sh first.")
endif()
set(CMAKE_MODULE_PATH ${COCOS2DX_ROOT_PATH}/cmake/Modules/)
include(CocosBuildSet)
add_subdirectory(${COCOS2DX_ROOT_PATH}/cocos ${ENGINE_BINARY_PATH}/cocos/core)
set(APP_NAME MyGame)
set(GAME_SOURCE Classes/AppDelegate.cpp Classes/CrystalQuestScene.cpp)
set(GAME_HEADER Classes/AppDelegate.h Classes/CrystalQuestScene.h)
if(ANDROID)
  list(APPEND GAME_SOURCE proj.android/app/jni/hellocpp/main.cpp)
  add_library(${APP_NAME} SHARED ${GAME_SOURCE} ${GAME_HEADER})
  add_subdirectory(${COCOS2DX_ROOT_PATH}/cocos/platform/android ${ENGINE_BINARY_PATH}/cocos/platform)
  target_link_libraries(${APP_NAME} -Wl,--whole-archive cpp_android_spec -Wl,--no-whole-archive)
else()
  add_executable(${APP_NAME} ${GAME_SOURCE} ${GAME_HEADER})
endif()
target_include_directories(${APP_NAME} PRIVATE Classes ../include ${COCOS2DX_ROOT_PATH}/cocos/audio/include)
target_link_libraries(${APP_NAME} cocos2d)
target_compile_features(${APP_NAME} PRIVATE cxx_std_17)
setup_cocos_app_config(${APP_NAME})
